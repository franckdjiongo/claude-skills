# Script template

Use this template structure for all governance scripts:

```powershell
<#
.SYNOPSIS
    Brief description of what the script does

.DESCRIPTION
    Detailed description of the script's purpose and behavior

.PARAMETER ParameterName
    Description of each parameter

.EXAMPLE
    Example-GovernanceScript -Parameter Value
    Description of what this example does

.NOTES
    Author: Generated by Power Platform Governance Script Generator
    Date: [Date]
    Version: 1.0
    Prerequisites:
        - PowerShell 5.1+
        - Microsoft.PowerApps.Administration.PowerShell module
        - Power Platform Administrator or Dynamics 365 Administrator role
        - First-time sign-in to Power Platform admin center completed

.LINK
    https://learn.microsoft.com/power-platform/admin/
#>

[CmdletBinding(SupportsShouldProcess)]
param(
    [Parameter(Mandatory=$false)]
    [switch]$DryRun,
    
    [Parameter(Mandatory=$false)]
    [string]$OutputPath = ".\reports",
    
    [Parameter(Mandatory=$false)]
    [string]$LogPath = ".\logs\governance-script.log"
)

#Requires -Version 5.1
#Requires -Modules Microsoft.PowerApps.Administration.PowerShell

# Initialize script
$ErrorActionPreference = "Stop"
$script:StartTime = Get-Date

# Logging function
function Write-Log {
    param(
        [string]$Message,
        [ValidateSet("Info", "Warning", "Error", "Success")]
        [string]$Level = "Info"
    )
    
    $timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
    $logEntry = "$timestamp [$Level] $Message"
    
    # Write to log file
    if ($LogPath) {
        $logDir = Split-Path -Path $LogPath -Parent
        if (-not (Test-Path $logDir)) {
            New-Item -ItemType Directory -Path $logDir -Force | Out-Null
        }
        Add-Content -Path $LogPath -Value $logEntry
    }
    
    # Write to console with color
    $color = switch ($Level) {
        "Info" { "White" }
        "Warning" { "Yellow" }
        "Error" { "Red" }
        "Success" { "Green" }
    }
    
    Write-Host $logEntry -ForegroundColor $color
}

# Error handling wrapper
function Invoke-WithRetry {
    param(
        [ScriptBlock]$ScriptBlock,
        [int]$MaxRetries = 3,
        [int]$RetryDelaySeconds = 5
    )
    
    $attempt = 1
    
    while ($attempt -le $MaxRetries) {
        try {
            return & $ScriptBlock
        }
        catch {
            if ($attempt -eq $MaxRetries) {
                throw
            }
            
            Write-Log "Attempt $attempt failed: $($_.Exception.Message). Retrying in $RetryDelaySeconds seconds..." -Level Warning
            Start-Sleep -Seconds $RetryDelaySeconds
            $attempt++
        }
    }
}

# Main function
function Invoke-GovernanceScript {
    try {
        Write-Log "=== Script Started ===" -Level Info
        Write-Log "Mode: $(if ($DryRun) { 'DRY RUN' } else { 'LIVE' })" -Level Info
        
        # Authenticate
        Write-Log "Authenticating to Power Platform..." -Level Info
        try {
            Add-PowerAppsAccount -ErrorAction Stop
            Write-Log "Authentication successful" -Level Success
        }
        catch {
            Write-Log "Authentication failed: $($_.Exception.Message)" -Level Error
            throw
        }
        
        # Main governance logic here
        Write-Log "Executing governance tasks..." -Level Info
        
        # Example: Get environments
        $environments = Invoke-WithRetry -ScriptBlock {
            Get-AdminPowerAppEnvironment
        }
        
        Write-Log "Found $($environments.Count) environments" -Level Info
        
        # Process each environment
        foreach ($env in $environments) {
            Write-Log "Processing environment: $($env.DisplayName)" -Level Info
            
            if ($PSCmdlet.ShouldProcess($env.DisplayName, "Process environment")) {
                if (-not $DryRun) {
                    # Actual operations here
                }
                else {
                    Write-Log "  [DRY RUN] Would process environment" -Level Warning
                }
            }
        }
        
        Write-Log "=== Script Completed Successfully ===" -Level Success
        
        # Calculate duration
        $duration = (Get-Date) - $script:StartTime
        Write-Log "Total duration: $($duration.ToString('hh\:mm\:ss'))" -Level Info
    }
    catch {
        Write-Log "Script failed: $($_.Exception.Message)" -Level Error
        Write-Log "Stack trace: $($_.ScriptStackTrace)" -Level Error
        throw
    }
}

# Execute
Invoke-GovernanceScript
```
